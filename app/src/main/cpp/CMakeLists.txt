cmake_minimum_required(VERSION 3.22.1)
project(atlas LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "CPU remains the validated fallback" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "llama.cpp em falta. Executa: bash scripts/setup-native.sh")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(atlas SHARED atlas.cpp)
target_link_libraries(atlas PRIVATE llama android log)
# Android's debug APK must not turn the inference kernels into an unoptimized build.
# Target options follow CMAKE_*_FLAGS_DEBUG, so -O3 overrides the default -O0.
foreach(inference_target atlas llama ggml ggml-base ggml-cpu)
    if(TARGET ${inference_target})
        target_compile_options(${inference_target} PRIVATE -O3)
        target_compile_definitions(${inference_target} PRIVATE NDEBUG)
    endif()
endforeach()
# Support devices using 16 KB memory pages, including Android 15+.
target_link_options(atlas PRIVATE "-Wl,-z,max-page-size=16384")
